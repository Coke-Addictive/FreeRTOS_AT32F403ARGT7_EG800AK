#include "eg_network.h"

#include <stdio.h>

#define LOG_TAG "EG_NET"
#include "log.h"

#define EG800_NETWORK_CONFIG_TIMEOUT_MS             5000U               // QICSGP配置超时
#define EG800_NETWORK_ACTIVATE_TIMEOUT_MS           60000U              // QIACT激活超时
#define EG800_NETWORK_QUERY_TIMEOUT_MS              5000U               // QIACT查询超时
#define EG800_NETWORK_DEACTIVATE_TIMEOUT_MS         40000U              // QIDEACT去激活超时

#define EG800_NETWORK_CONTEXT_ID_MAX                15U                 // TCP/IP PDP 场景 ID 上限
#define EG800_NETWORK_APN_MAX_LENGTH                50U                 // 当前封装允许的 APN 最大长度，单位字节
#define EG800_NETWORK_QUERY_RESPONSE_SIZE           512U                // QIACT 查询响应缓冲区容量，单位字节

// Network层内部AT命令编号
typedef enum {
    EG800_NETWORK_CMD_QICSGP = 0U,              // 配置PDP场景和APN
    EG800_NETWORK_CMD_QIACT,                    // 激活PDP场景
    EG800_NETWORK_CMD_QIACT_QUERY,              // 查询PDP场景状态和IP
    EG800_NETWORK_CMD_QIDEACT                   // 去激活PDP场景
} Eg800NetworkCmd_e;

// 单个运营商对应的APN配置
typedef struct {
    Eg800BasicOperator_e operator_type;         // 运营商类型
    const char *apn;                            // 运营商对应的APN
    uint8_t auth_type;                          // QICSGP鉴权方式
} Eg800NetworkApnConfig_t;

static TaskHandle_t s_eg800_network_task_handle = NULL;                 // 任务句柄

static bool s_eg800_network_initialized = false;                        // 初始化标志位

static bool s_eg800_network_should_connect = false;                     // 上层是否希望网络保持连接（目标状态）

static uint8_t s_eg800_network_retry_count = 0U;                        // 当前联网流程已经尝试的次数

static Eg800NetworkSnapshot_t s_eg800_network_snapshot = {0};           // Network 信息缓存及刷新状态

static SemaphoreHandle_t s_network_at_completion_sem = NULL;            // Network 层 AT 请求完成信号量

// 实际处理URC的函数声明
static void Eg800_Network_TCPIP_Urc_Handler(const char *line);

// 路由表需要在声明之后
static const Eg800UrcRegistration_t s_eg800_network_urc_routes[] = {
    {"+QIURC:", EG800_AT_OWNER_NETWORK, Eg800_Network_TCPIP_Urc_Handler},
};

// 当前支持的运营商APN表
static const Eg800NetworkApnConfig_t s_eg800_network_apn_configs[] = {
    {EG800_BASIC_OPERATOR_CHINA_MOBILE,     "CMNET",    0U},
    {EG800_BASIC_OPERATOR_CHINA_TELECOM,    "CTNET",    0U},
    {EG800_BASIC_OPERATOR_CHINA_UNICOM,     "UNINET",   0U}
};

static void Eg800_Network_Info_Reset(void);

static const Eg800NetworkApnConfig_t *Eg800_Network_Find_Apn(Eg800BasicOperator_e operator_type);

//--------------------------------------------------------URC 处理-----------------------------------------
static void Eg800_Network_TCPIP_Urc_Handler(const char *line) {
    LOG_ERROR("%s", line);
    LOG("收到PDP去激活通知或者说是断网通知");
}

//--------------------------------------------------------关键函数实体-----------------------------------------
/**
 * @brief 将 Network 层保存的信息恢复为未知状态。
 */
static void Eg800_Network_Info_Reset(void) {
    (void)memset(&s_eg800_network_snapshot, 0, sizeof(s_eg800_network_snapshot));
    s_eg800_network_snapshot.info.state = EG800_NETWORK_STATE_NONE;
    s_eg800_network_snapshot.info.pdp_id = EG800_NETWORK_CONTEXT_ID;
    s_eg800_network_snapshot.info.pdp_type = EG800_NETWORK_CONTEXT_TYPE;
}

/**
 * @brief 注册 Network 层负责的全部主动上报 URC。
 * @return RESULT_SUCCESS 表示全部注册成功，RESULT_FAIL 表示存在注册失败。
 */
static Result_t Eg800_Network_Register_Urcs(void) {
    return Eg800_Urc_Register_Table(s_eg800_network_urc_routes, ARRAY_SIZE(s_eg800_network_urc_routes));
}



/**
 * @brief 同步执行一条 Network 服务的普通文本 AT 命令。
 * @param cmd_id Network 服务内部命令编号。
 * @param name 命令日志名称，可为 NULL。
 * @param tx 完整 AT 发送字符串，必须包含 "\r\n"，调用期间保持有效。
 * @param rsp_prefix 需要保存的响应前缀，NULL 表示保存未被 URC 路由消费的文本。
 * @param rx_buf 响应接收缓冲区，调用期间保持有效；不需要响应内容时传 NULL。
 * @param rx_buf_size 响应缓冲区容量，单位字节；rx_buf 为 NULL 时必须为 0。
 * @param timeout_ms 等待 OK 或 ERROR 的超时时间，单位 ms；0 表示使用 Service 默认值。
 * @param result_info AT 执行详细结果，可为 NULL。
 * @param completion_sem 当前调用任务专用的二值信号量，调用期间保持有效。
 * @return EG800_AT_RESULT_OK 表示命令成功，其他值表示对应的执行错误。
 * @note Network 和 AT 核心服务应已初始化；仅供任务上下文串行调用。
 * @note 禁止在 ISR、AT 核心 Service 任务及其 URC 回调中调用。
 * @note 同一信号量同一时间只允许对应一条未完成请求。
 */
static Eg800AtResult_e Eg800_Network_At_Execute(Eg800NetworkCmd_e cmd_id, const char *name, const char *tx, const char *rsp_prefix, char *rx_buf, size_t rx_buf_size, uint32_t timeout_ms, Eg800AtResultInfo_t *result_info, SemaphoreHandle_t completion_sem) {
    Eg800AtRequest_t request = {0};

    request.cmd.owner = EG800_AT_OWNER_NETWORK;
    request.cmd.id = (uint8_t)cmd_id;
    request.flow = EG800_AT_FLOW_TEXT;

    request.name = name;
    request.tx = tx;
    request.rsp_prefix = rsp_prefix;
    request.rx_buf = rx_buf;
    request.rx_buf_size = rx_buf_size;

    request.final_timeout_ms = timeout_ms;

    return Eg800_AT_Execute(&request, result_info, completion_sem);
}

/**
 * @brief 根据 Basic 缓存执行一次 PDP 配置、激活和 IP 查询。
 * @param info 本次联网结果输出；参数检查通过后先复位，仅返回 OK 时可作为联网成功结果使用。
 * @return OK 表示目标 PDP 已激活且返回了非空、非 0.0.0.0 的 IPv4 地址；CANCELLED 表示联网意图已关闭。
 * @return PARAM_ERROR 表示未初始化或参数无效；DATA_FAIL 表示前置条件或查询结果不满足要求；AT 失败时返回原错误码。
 * @note 仅供 Network 任务串行调用，使用 Network 专用 AT 完成信号量。
 * @note 调用前应确认目标 PDP 未激活；本函数不更新全局快照，不执行重试、CFUN 恢复或 PDP 清理。
 * @note 返回失败或取消时，PDP 仍可能已经激活，需要调用任务处理。
 */
static Eg800AtResult_e Eg800_Network_Connect_Once(Eg800NetworkInfo_t *info) {
    Eg800BasicSnapshot_t basic_snapshot;
    Eg800NetworkPdpContext_t pdp_context;
    const Eg800NetworkApnConfig_t *apn_config;
    Eg800AtResult_e result;

    if ((s_eg800_network_initialized == false) || (info == NULL) || (s_network_at_completion_sem == NULL)) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    // 先清除上一次调用留下的结果，避免失败时沿用旧 IP。
    (void)memset(info, 0, sizeof(*info));
    info->pdp_id = EG800_NETWORK_CONTEXT_ID;
    info->pdp_type = EG800_NETWORK_CONTEXT_TYPE;
    info->state = EG800_NETWORK_STATE_NONE;
    info->operator_type = EG800_BASIC_OPERATOR_UNKNOWN;

    if (Eg800_Network_Should_Connect() == false) {
        return EG800_AT_RESULT_CANCELLED;
    }

    // 获取当前缓存，本次调用不等待 Basic 后台刷新完成。
    if (Eg800_Basic_Get_Snapshot(&basic_snapshot) != RESULT_SUCCESS) {
        LOG_ERROR("单次联网读取 Basic 快照失败");
        return EG800_AT_RESULT_DATA_FAIL;
    }

    if (basic_snapshot.info.sim_state != EG800_BASIC_SIM_STATE_READY) {
        LOG_WARN("单次联网 SIM 未就绪: sim_state=%d", basic_snapshot.info.sim_state);
        return EG800_AT_RESULT_DATA_FAIL;
    }

    apn_config = Eg800_Network_Find_Apn(basic_snapshot.info.operator_type);
    if (apn_config == NULL) {
        LOG_WARN("单次联网未找到 APN: operator_type=%d", basic_snapshot.info.operator_type);
        return EG800_AT_RESULT_DATA_FAIL;
    }

    // 前置检查期间也可能收到取消请求，发送第一条 AT 前再次确认。
    if (Eg800_Network_Should_Connect() == false) {
        return EG800_AT_RESULT_CANCELLED;
    }

    LOG("单次联网开始: context_id=%d, context_type=%d, apn=%s", EG800_NETWORK_CONTEXT_ID, EG800_NETWORK_CONTEXT_TYPE, apn_config->apn);

    result = Eg800_Network_Configure_PDP_Context(EG800_NETWORK_CONTEXT_ID, EG800_NETWORK_CONTEXT_TYPE, apn_config->apn, s_network_at_completion_sem);

    // 同步等待 AT 期间，上层可能取消联网；返回后优先处理取消。
    if (Eg800_Network_Should_Connect() == false) {
        return EG800_AT_RESULT_CANCELLED;
    }

    if (result != EG800_AT_RESULT_OK) {
        LOG_ERROR("单次联网配置 PDP 失败: result=%d", result);
        return result;
    }

    result = Eg800_Network_Activate_PDP(EG800_NETWORK_CONTEXT_ID, s_network_at_completion_sem);
    if (result != EG800_AT_RESULT_OK) {
        LOG_ERROR("单次联网激活 PDP 失败: result=%d", result);
        return result;
    }

    result = Eg800_Network_Query_PDP_Context(EG800_NETWORK_CONTEXT_ID, &pdp_context, s_network_at_completion_sem);
    if (result != EG800_AT_RESULT_OK) {
        LOG_ERROR("单次联网查询 PDP 失败: result=%d", result);
        return result;
    }

    // 查询成功不等于目标 PDP 已激活，还需要检查具体结果。
    if ((pdp_context.context_id != EG800_NETWORK_CONTEXT_ID) || (pdp_context.active == false) || (pdp_context.context_type != EG800_NETWORK_CONTEXT_TYPE)) {
        LOG_ERROR("单次联网 PDP 状态不满足要求: context_id=%u, active=%u, context_type=%u", (unsigned int)pdp_context.context_id, (unsigned int)pdp_context.active, (unsigned int)pdp_context.context_type);
        return EG800_AT_RESULT_DATA_FAIL;
    }

    if ((pdp_context.ip_addr[0] == '\0') || (strcmp(pdp_context.ip_addr, "0.0.0.0") == 0)) {
        LOG_ERROR("单次联网未取得 IP 地址: ip=%s", pdp_context.ip_addr);
        return EG800_AT_RESULT_DATA_FAIL;
    }

    // 只填写调用者的本次结果，由 Network 任务决定是否发布为全局快照。
    info->operator_type = apn_config->operator_type;
    (void)memcpy(info->ip_addr, pdp_context.ip_addr, sizeof(info->ip_addr));
    info->state = EG800_NETWORK_STATE_ACTIVE;

    return EG800_AT_RESULT_OK;
}



//-------------------------------------------------其他----------------------------------------------
/**
 * @brief 根据运营商查找当前支持的 APN 配置。
 * @param operator_type Basic 层提供的运营商类型。
 * @return 找到时返回只读配置表项的地址，整个程序运行期间有效；未知或不支持时返回 NULL。
 */
static const Eg800NetworkApnConfig_t *Eg800_Network_Find_Apn(Eg800BasicOperator_e operator_type) {
    size_t index;

    for (index = 0U; index < ARRAY_SIZE(s_eg800_network_apn_configs); index++) {
        if (s_eg800_network_apn_configs[index].operator_type == operator_type) {
            return &s_eg800_network_apn_configs[index];
        }
    }

    return NULL;
}

/**
 * @brief 在临界区内读取上层的联网意图。
 * @return true 表示仍希望保持联网，false 表示当前不应继续联网流程。
 */
static bool Eg800_Network_Should_Connect(void) {
    bool should_connect;

    taskENTER_CRITICAL();
    should_connect = s_eg800_network_should_connect;
    taskEXIT_CRITICAL();

    return should_connect;
}



/**
 * @brief 更新 Network 全局快照，并清除非联网状态下的 IP 地址。
 * @param info 当前状态及 PDP 信息，调用期间保持有效；发布 ACTIVE 时必须已确认联网成功。
 * @note 仅由 Network 任务调用；参数为空时记录错误并保持原快照。
 * @note 本函数只发布软件状态，实际联网、去激活和恢复由调用者安排。
 */
static void Eg800_Network_Update_Snapshot(const Eg800NetworkInfo_t *info) {
    TickType_t update_tick;

    if (info == NULL) {
        LOG_ERROR("Network 快照更新参数为空");
        return;
    }

    update_tick = xTaskGetTickCount();

    taskENTER_CRITICAL();

    s_eg800_network_snapshot.info = *info;
    s_eg800_network_snapshot.update_tick = update_tick;
    s_eg800_network_snapshot.refresh_in_progress = (info->state == EG800_NETWORK_STATE_CONNECTING) || (info->state == EG800_NETWORK_STATE_DEACTIVATING);

    // 离开 ACTIVE 后，旧 IP 不再作为当前可用地址对外提供。
    if (info->state != EG800_NETWORK_STATE_ACTIVE) {
        (void)memset(s_eg800_network_snapshot.info.ip_addr, 0, sizeof(s_eg800_network_snapshot.info.ip_addr));
    }

    taskEXIT_CRITICAL();
}


















//--------------------------------------------------------公开的API接口--------------------------------------------------------
/**
 * @brief 初始化 Network 层并注册基础 URC。
 * @return RESULT_SUCCESS 表示初始化成功。
 * @note 本函数不发送 AT 命令，可以在模块上电前调用。
 */
Result_t Eg800_Network_Init(void) {
    BaseType_t task_result;

    // 幂等检查
    if (s_eg800_network_initialized == true) {
        return RESULT_SUCCESS;
    }

    Eg800_Network_Info_Reset();

    // 注册URC
    if (Eg800_Network_Register_Urcs() != RESULT_SUCCESS) {
        return RESULT_FAIL;
    }

    // 创建network 信号量
    s_network_at_completion_sem = xSemaphoreCreateBinary();
    if (s_network_at_completion_sem == NULL) {
        LOG_ERROR("Network AT 完成信号量创建失败");
        return RESULT_FAIL;
    }

    task_result = xTaskCreate(Eg800_Network_Task,
                              "EG Network Core Task",
                              EG800_NETWORK_TASK_STACK_SIZE,
                              NULL,
                              EG800_NETWORK_TASK_PRIORITY,
                              &s_eg800_network_task_handle);

    if (task_result != pdPASS) {
        s_eg800_network_task_handle = NULL;
        LOG_ERROR("Network 后台刷新任务创建失败");
        return RESULT_FAIL;
    }

    s_eg800_network_initialized = true;

    return RESULT_SUCCESS;
}




//----------------------------------------------------基础 AT 命令封装----------------------------------------------------
/**
 * @brief 配置指定 PDP 场景的协议类型和 APN，使用空用户名、空密码及无鉴权方式。
 * @param context_id PDP 场景 ID，范围 1~15。
 * @param context_type 协议类型，1 为 IPv4，2 为 IPv6，3 为 IPv4v6。
 * @param apn APN 字符串，最多 50 字节；允许空字符串，不允许引号、反斜杠及控制字符。
 * @param completion_sem 当前调用任务专用的 AT 完成信号量，同一时间仅供一条请求使用。
 * @return AT 执行结果；未初始化、参数无效或命令生成失败时返回 PARAM_ERROR。
 * @note 本接口只配置 PDP 参数，不激活网络，也不修改联网意图及 Network 快照。
 */
Eg800AtResult_e Eg800_Network_Configure_PDP_Context(uint8_t context_id, uint8_t context_type, const char *apn, SemaphoreHandle_t completion_sem) {
    size_t apn_len;
    char ch;
    int command_len;
    char command[EG800_NETWORK_APN_MAX_LENGTH + 40U];

    // 合法性检查
    if ((s_eg800_network_initialized == false) || (context_id == 0U) || (context_id > EG800_NETWORK_CONTEXT_ID_MAX) || (context_type < 1U) || (context_type > 3U) || (apn == NULL) || (completion_sem == NULL)) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    // APN 会直接写入带引号的 AT 参数，禁止破坏参数边界或引入额外命令。
    for (apn_len = 0U; apn_len <= EG800_NETWORK_APN_MAX_LENGTH; apn_len++) {
        ch = apn[apn_len];
        if (ch == '\0') {
            break;
        }
        if ((ch < 0x20U) || (ch > 0x7EU) || (ch == '"') || (ch == '\\')) {
            return EG800_AT_RESULT_PARAM_ERROR;
        }
    }

    if (apn_len > EG800_NETWORK_APN_MAX_LENGTH) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    command_len = snprintf(command, sizeof(command), "AT+QICSGP=%u,%u,\"%s\",\"\",\"\",0\r\n", (unsigned int)context_id, (unsigned int)context_type, apn);
    if ((command_len <= 0) || ((size_t)command_len >= sizeof(command))) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    return Eg800_Network_At_Execute(EG800_NETWORK_CMD_QICSGP, "配置 PDP 上下文场景", command, NULL, NULL, 0U, EG800_NETWORK_CONFIG_TIMEOUT_MS, NULL, completion_sem);
}


/**
 * @brief 激活指定 PDP 场景。
 * @param context_id PDP 场景 ID，范围 1~15。
 * @param completion_sem 当前调用任务专用的 AT 完成信号量，同一时间仅供一条请求使用。
 * @return AT 执行结果；未初始化、参数无效或命令生成失败时返回 PARAM_ERROR。
 * @note OK 仅表示激活命令成功；IP 地址应通过 Query_PDP_Context 查询，不在此更新快照。
 */
Eg800AtResult_e Eg800_Network_Activate_PDP(uint8_t context_id, SemaphoreHandle_t completion_sem) {
    char command[24];
    int command_len;

    // 合法性检查
    if ((s_eg800_network_initialized == false) || (context_id == 0U) || (context_id > EG800_NETWORK_CONTEXT_ID_MAX) || (completion_sem == NULL)) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    command_len = snprintf(command, sizeof(command), "AT+QIACT=%u\r\n", (unsigned int)context_id);
    if ((command_len <= 0) || ((size_t)command_len >= sizeof(command))) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    return Eg800_Network_At_Execute(EG800_NETWORK_CMD_QIACT, "激活 PDP 场景", command, NULL, NULL, 0U, EG800_NETWORK_ACTIVATE_TIMEOUT_MS, NULL, completion_sem);
}


/**
 * @brief 查询 PDP 激活列表，提取指定场景的激活状态、协议类型和 IP 地址。
 * @param context_id 需要查询的 PDP 场景 ID，范围 1~15。
 * @param context 查询结果输出；参数检查通过后先清空，只有返回 OK 时才可使用。
 * @param completion_sem 当前调用任务专用的 AT 完成信号量，同一时间仅供一条请求使用。
 * @return AT 执行结果；参数无效时返回 PARAM_ERROR，响应截断或格式错误时返回 DATA_FAIL。
 * @note 查询成功但列表中没有指定 ID 时，返回 OK，active 为 false，context_type 为 0，IP 为空。
 * @note 结构体只保存一个地址；同一场景同时返回 IPv4 和 IPv6 时优先保存 IPv4。
 * @note 本接口不修改 Network 快照；响应缓冲区为 512 字节的普通局部数组，调用任务必须预留足够的栈空间。
 */
Eg800AtResult_e Eg800_Network_Query_PDP_Context(uint8_t context_id, Eg800NetworkPdpContext_t *context, SemaphoreHandle_t completion_sem) {
    char response[EG800_NETWORK_QUERY_RESPONSE_SIZE] = {0}; // 整块相应缓冲区
    char *line;                                         //当前正在解析的这一行的首地址
    char *next_line;                                    //当前行末尾的 '\n' 地址；加 1 才是下一行起点
    char *ip_start;                                     //解析过程中逐步移动，最终指向 IP 的第一个字符
    char *ip_end;                                       //IP 后面那个结束双引号的地址
    char *tail;                                         //结束双引号后面的地址，用于检查行尾
    size_t ip_len;              
    unsigned int parsed_id;                             //当前这一行解析出来的场景 ID
    unsigned int parsed_state;                          //当前这一行的激活状态
    unsigned int parsed_type;                           //当前这一行的地址类型
    int fields_end;                                     //sscanf() 解析完三个数值字段后，已经读取的字符数
    Eg800NetworkPdpContext_t parsed_context = {0};      //暂存准备返回给调用者的结果
    Eg800AtResultInfo_t result_info = {0};              //AT 核心返回的详细执行信息，这里主要使用 rx_len
    Eg800AtResult_e result;

    // 合法性检查
    if ((s_eg800_network_initialized == false) || (context_id == 0U) || (context_id > EG800_NETWORK_CONTEXT_ID_MAX) || (context == NULL) || (completion_sem == NULL)) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    // 清空缓冲区
    (void)memset(context, 0, sizeof(*context));

    context->context_id = context_id;
    parsed_context.context_id = context_id;


    result = Eg800_Network_At_Execute(EG800_NETWORK_CMD_QIACT_QUERY, "查询 PDP 场景", "AT+QIACT?\r\n", "+QIACT:", response, EG800_NETWORK_QUERY_RESPONSE_SIZE, EG800_NETWORK_QUERY_TIMEOUT_MS, &result_info, completion_sem);
    if (result != EG800_AT_RESULT_OK) {
        return result;
    }

    // Service 在缓冲区不足时截断文本，不能据此认定未出现在响应中的场景没有激活。
    if (result_info.rx_len >= (EG800_NETWORK_QUERY_RESPONSE_SIZE - 1U)) {
        return EG800_AT_RESULT_DATA_FAIL;
    }

    line = response;
    while (*line != '\0') {
        next_line = strchr(line, '\n');
        if (next_line != NULL) {
            *next_line = '\0';
        }

        // 限制数值字段宽度，并用 %n 确认三个字段均已完整匹配。
        fields_end = 0;
        if ((sscanf(line, "+QIACT: %2u , %1u , %1u %n", &parsed_id, &parsed_state, &parsed_type, &fields_end) != 3) || (fields_end == 0) || (parsed_id == 0U) || (parsed_id > EG800_NETWORK_CONTEXT_ID_MAX) || (parsed_state > 1U) || (parsed_type < 1U) || (parsed_type > 2U)) {
            result = EG800_AT_RESULT_DATA_FAIL;
            break;
        }

        ip_start = line + fields_end;
        ip_len = 0U;
        if (*ip_start != '\0') {
            if (*ip_start != ',') {
                result = EG800_AT_RESULT_DATA_FAIL;
                break;
            }
            ip_start++;
            while ((*ip_start == ' ') || (*ip_start == '\t')) {
                ip_start++;
            }
            if (*ip_start != '"') {
                result = EG800_AT_RESULT_DATA_FAIL;
                break;
            }
            ip_start++;
            ip_end = strchr(ip_start, '"');
            if (ip_end == NULL) {
                result = EG800_AT_RESULT_DATA_FAIL;
                break;
            }
            ip_len = (size_t)(ip_end - ip_start);
            tail = ip_end + 1;
            while ((*tail == ' ') || (*tail == '\t') || (*tail == '\r')) {
                tail++;
            }
            if ((ip_len >= sizeof(parsed_context.ip_addr)) || (*tail != '\0')) {
                result = EG800_AT_RESULT_DATA_FAIL;
                break;
            }
        }

        if ((parsed_state == 1U) && (ip_len == 0U)) {
            result = EG800_AT_RESULT_DATA_FAIL;
            break;
        }

        // 必须解析完整列表；后续响应损坏时，不能把前面已找到的数据当成有效查询结果。
        if ((parsed_id == context_id) && ((parsed_context.active == false) || ((parsed_context.context_type == 2U) && (parsed_type == 1U) && (parsed_state == 1U)))) {
            parsed_context.active = (parsed_state == 1U);
            parsed_context.context_type = (uint8_t)parsed_type;
            (void)memset(parsed_context.ip_addr, 0, sizeof(parsed_context.ip_addr));
            if (parsed_context.active == true) {
                (void)memcpy(parsed_context.ip_addr, ip_start, ip_len);
            }
        }

        if (next_line == NULL) {
            break;
        }
        line = next_line + 1;
    }

    if (result == EG800_AT_RESULT_OK) {
        *context = parsed_context;
    }
    return result;
}

/**
 * @brief 去激活指定 PDP 场景，同时关闭依附该场景的 TCP/IP 连接。
 * @param context_id PDP 场景 ID，范围 1~15。
 * @param completion_sem 当前调用任务专用的 AT 完成信号量，同一时间仅供一条请求使用。
 * @return AT 执行结果；未初始化、参数无效或命令生成失败时返回 PARAM_ERROR。
 * @note 本接口只执行去激活命令；停止自动重连应由 Disconnect 设置联网意图。
 */
Eg800AtResult_e Eg800_Network_Deactivate_PDP(uint8_t context_id, SemaphoreHandle_t completion_sem) {
    char command[24];
    int command_len;

    if ((s_eg800_network_initialized == false) || (context_id == 0U) || (context_id > EG800_NETWORK_CONTEXT_ID_MAX) || (completion_sem == NULL)) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    command_len = snprintf(command, sizeof(command), "AT+QIDEACT=%u\r\n", (unsigned int)context_id);
    if ((command_len <= 0) || ((size_t)command_len >= sizeof(command))) {
        return EG800_AT_RESULT_PARAM_ERROR;
    }

    return Eg800_Network_At_Execute(EG800_NETWORK_CMD_QIDEACT, "去激活 PDP 场景", command, NULL, NULL, 0U, EG800_NETWORK_DEACTIVATE_TIMEOUT_MS, NULL, completion_sem);
}












/**
 * @brief 提交建立并保持 PDP 网络连接的请求。
 * @return RESULT_SUCCESS 表示请求已提交，不代表网络已经连接；RESULT_FAIL 表示 Network 未初始化或任务不存在。
 * @note 仅供任务上下文调用；实际联网由 Network 任务执行。
 * @note 本接口不重置重试次数，重复请求的处理由 Network 任务决定。
 */
Result_t Eg800_Network_Connect(void) {
    if ((s_eg800_network_initialized == false) || (s_eg800_network_task_handle == NULL)) {
        return RESULT_FAIL;
    }

    // 先更新联网意图，确保任务被唤醒后能够读到目标状态。
    taskENTER_CRITICAL();
    s_eg800_network_should_connect = true;
    taskEXIT_CRITICAL();

    (void)xTaskNotify(s_eg800_network_task_handle, EG800_NETWORK_EVENT_CONNECT, eSetBits);

    return RESULT_SUCCESS;
}




/**
 * @brief 关闭保持联网的意图，并提交 PDP 断开请求。
 * @return RESULT_SUCCESS 表示请求已提交，不代表 PDP 已经断开；RESULT_FAIL 表示 Network 未初始化或任务不存在。
 * @note 仅供任务上下文调用；实际去激活和清理由 Network 任务执行。
 * @note 已进入激活阶段时，由 Network 任务等待当前联网流程返回后处理断开。
 */
Result_t Eg800_Network_Disconnect(void) {
    if ((s_eg800_network_initialized == false) || (s_eg800_network_task_handle == NULL)) {
        return RESULT_FAIL;
    }

    // 先关闭联网意图，任务处理请求时据此停止后续联网尝试。
    taskENTER_CRITICAL();
    s_eg800_network_should_connect = false;
    taskEXIT_CRITICAL();

    (void)xTaskNotify(s_eg800_network_task_handle, EG800_NETWORK_EVENT_DISCONNECT, eSetBits);

    return RESULT_SUCCESS;
}




/**
 * @brief 判断PDP网络当前是否正常可用。
 * @return true表示PDP已激活并已经获得IP地址。
 */
bool Eg800_Network_Is_Active(void) {
    bool active;

    if (s_eg800_network_initialized == false) {
        return false;
    }

    taskENTER_CRITICAL();
    active = (s_eg800_network_snapshot.info.state == EG800_NETWORK_STATE_ACTIVE ? true : false);
    taskEXIT_CRITICAL();

    return active;
}




/**
 * @brief 获取 Network 当前保存的状态快照。
 * @param snapshot 快照输出地址；返回成功时有效，返回失败时不修改其内容。
 * @return RESULT_SUCCESS 表示快照复制成功，不代表网络已连接；RESULT_FAIL 表示未初始化或输出地址为空。
 * @note 仅供任务上下文调用，读取软件缓存，不发送 AT 命令。
 */
Result_t Eg800_Network_Get_Snapshot(Eg800NetworkSnapshot_t *snapshot) {
    if ((s_eg800_network_initialized == false) || (snapshot == NULL)) {
        return RESULT_FAIL;
    }

    taskENTER_CRITICAL();
    *snapshot = s_eg800_network_snapshot;
    taskEXIT_CRITICAL();

    return RESULT_SUCCESS;
}

























//-----------------------------------------------网络层核心维护任务-------------------------------------------------------
/**
 * @brief 初始化 Network 维护信息，并接收上层控制请求。
 * @param argument 任务创建参数，当前未使用。
 * @note 当前阶段已接入单次联网和 PDP 清理，定时重试及掉线恢复后续加入。
 */
static void Eg800_Network_Task(void *argument) {
    Eg800AtResult_e result;
    Eg800AtResult_e cleanup_result;
    uint32_t events;
    BaseType_t wait_result;
    TickType_t wait_ticks = portMAX_DELAY;
    bool should_connect;
    Eg800NetworkInfo_t network_info = {0};
    TickType_t retry_start_tick = 0U;
    TickType_t retry_elapsed_ticks;
    const TickType_t retry_interval_ticks = pdMS_TO_TICKS(EG800_NETWORK_RETRY_INTERVAL_MS);

    (void)argument;

    // 尚未确认模块中的 PDP 状态，初始阶段保持 NONE。
    network_info.pdp_id = EG800_NETWORK_CONTEXT_ID;
    network_info.pdp_type = EG800_NETWORK_CONTEXT_TYPE;
    network_info.state = EG800_NETWORK_STATE_NONE;
    network_info.operator_type = EG800_BASIC_OPERATOR_UNKNOWN;
    Eg800_Network_Update_Snapshot(&network_info);

    while (1) {
        events = 0U;
        wait_result = xTaskNotifyWait(0U, UINT32_MAX, &events, wait_ticks);

        if (wait_result == pdTRUE) {
            LOG("Network 收到事件: events=0x%08lX", (unsigned long)events);
        }

        // 事件可能合并，实际目标必须以当前保存的联网意图为准。
        should_connect = Eg800_Network_Should_Connect();

        if (should_connect == false) {
            // 已经不希望联网，停止等待联网重试期限。
            wait_ticks = portMAX_DELAY;

            // 这两个状态都已确认 PDP 清理成功，取消联网时直接结束本轮尝试。
            if ((network_info.state == EG800_NETWORK_STATE_IDLE) || (network_info.state == EG800_NETWORK_STATE_RETRY_WAIT)) {
                s_eg800_network_retry_count = 0U;
                network_info.state = EG800_NETWORK_STATE_IDLE;
                Eg800_Network_Update_Snapshot(&network_info);
                continue;
            }

            // 故障状态下，只有收到新的明确断网请求，才再次尝试清理 PDP。
            if ((network_info.state == EG800_NETWORK_STATE_FAULT) && ((events & EG800_NETWORK_EVENT_DISCONNECT) == 0U)) {
                continue;
            }

            network_info.state = EG800_NETWORK_STATE_DEACTIVATING;
            (void)memset(network_info.ip_addr, 0, sizeof(network_info.ip_addr));
            Eg800_Network_Update_Snapshot(&network_info);

            LOG("Network 开始清理 PDP: context_id=%d", EG800_NETWORK_CONTEXT_ID);

            result = Eg800_Network_Deactivate_PDP(EG800_NETWORK_CONTEXT_ID, s_network_at_completion_sem);

            if (result == EG800_AT_RESULT_OK) {
                network_info.state = EG800_NETWORK_STATE_IDLE;
                LOG("Network PDP 清理成功");
            } else {
                network_info.state = EG800_NETWORK_STATE_FAULT;
                LOG_ERROR("Network PDP 清理失败: result=%d", (int)result);
            }

            Eg800_Network_Update_Snapshot(&network_info);
            continue;


        } else {


            // 已经联网时，重复 Connect 请求不重新执行联网流程。
            if (network_info.state == EG800_NETWORK_STATE_ACTIVE) {
                wait_ticks = portMAX_DELAY;
                continue;
            }

            // 故障停止后，重复 Connect 请求不能重新开启尝试或清零次数。
            if (network_info.state == EG800_NETWORK_STATE_FAULT) {
                wait_ticks = portMAX_DELAY;
                continue;
            }

            // 软件状态未知时，先清理模块中可能存在的 PDP。
            if (network_info.state == EG800_NETWORK_STATE_NONE) {
                network_info.state = EG800_NETWORK_STATE_DEACTIVATING;
                (void)memset(network_info.ip_addr, 0, sizeof(network_info.ip_addr));
                Eg800_Network_Update_Snapshot(&network_info);

                LOG("Network 联网前清理 PDP: context_id=%u", (unsigned int)EG800_NETWORK_CONTEXT_ID);

                result = Eg800_Network_Deactivate_PDP(EG800_NETWORK_CONTEXT_ID, s_network_at_completion_sem);

                if (result != EG800_AT_RESULT_OK) {
                    network_info.state = EG800_NETWORK_STATE_FAULT;
                    wait_ticks = portMAX_DELAY;
                    Eg800_Network_Update_Snapshot(&network_info);

                    LOG_ERROR("Network 联网前清理失败: result=%d", (int)result);
                    continue;
                }

                network_info.state = EG800_NETWORK_STATE_IDLE;
                Eg800_Network_Update_Snapshot(&network_info);
                LOG("Network 联网前清理成功");
            }

            // 通知可能提前唤醒任务，必须根据原始起点判断是否已经到达重试期限。
            if (network_info.state == EG800_NETWORK_STATE_RETRY_WAIT) {
                retry_elapsed_ticks = xTaskGetTickCount() - retry_start_tick;

                if (retry_elapsed_ticks < retry_interval_ticks) {
                    wait_ticks = retry_interval_ticks - retry_elapsed_ticks;
                    continue;
                }

                // 上次失败后已完成清理，等待结束后允许再次执行单次联网。
                network_info.state = EG800_NETWORK_STATE_IDLE;
            }

            if (network_info.state != EG800_NETWORK_STATE_IDLE) {
                continue;
            }

            // 清理期间可能收到断网请求，开始联网前再次确认意图。
            if (Eg800_Network_Should_Connect() == false) {
                continue;
            }

            LOG("Network 联网前准备完成");

            // 发布整个单次联网流程正在执行的状态。
            network_info.state = EG800_NETWORK_STATE_CONNECTING;
            Eg800_Network_Update_Snapshot(&network_info);

            s_eg800_network_retry_count++;
            LOG("Network 开始第 %u/%u 次联网", (unsigned int)s_eg800_network_retry_count, (unsigned int)EG800_NETWORK_RETRY_MAX);

            result = Eg800_Network_Connect_Once(&network_info);

            // 联网意图检查与成功快照发布一起完成，避免中间插入断网请求。
            taskENTER_CRITICAL();
            should_connect = s_eg800_network_should_connect;

            if ((result == EG800_AT_RESULT_OK) && (should_connect == true)) {
                Eg800_Network_Update_Snapshot(&network_info);
            }

            taskEXIT_CRITICAL();

            if ((result == EG800_AT_RESULT_OK) && (should_connect == true)) {
                s_eg800_network_retry_count = 0U;
                wait_ticks = portMAX_DELAY;
                LOG("Network 联网成功: ip=%s", network_info.ip_addr);
                continue;
            }

            // 联网失败、取消或上层已要求断开时，立即处理可能残留的 PDP。
            LOG("Network 本次流程结束，准备清理: result=%d, should_connect=%u", (int)result, (unsigned int)should_connect);

            network_info.state = EG800_NETWORK_STATE_DEACTIVATING;
            (void)memset(network_info.ip_addr, 0, sizeof(network_info.ip_addr));
            Eg800_Network_Update_Snapshot(&network_info);

            cleanup_result = Eg800_Network_Deactivate_PDP(EG800_NETWORK_CONTEXT_ID, s_network_at_completion_sem);

            if (cleanup_result != EG800_AT_RESULT_OK) {
                network_info.state = EG800_NETWORK_STATE_FAULT;
                wait_ticks = portMAX_DELAY;
                Eg800_Network_Update_Snapshot(&network_info);

                LOG_ERROR("Network 清理失败: connect_result=%d, cleanup_result=%d", (int)result, (int)cleanup_result);
                continue;
            }

            // 重新读取清理结束时的意图；should_connect 保留的是联网结束时的意图。
            if (Eg800_Network_Should_Connect() == false) {
                s_eg800_network_retry_count = 0U;
                network_info.state = EG800_NETWORK_STATE_IDLE;
                wait_ticks = portMAX_DELAY;
                LOG("Network 清理完成，保持断开");
            } else if (should_connect == false) {
                // 已观察到上层取消本轮联网，清理期间的新请求开启下一轮。
                s_eg800_network_retry_count = 0U;
                network_info.state = EG800_NETWORK_STATE_IDLE;
                wait_ticks = 0U;
                LOG("Network 清理完成，准备响应新的联网请求");
            } else if (s_eg800_network_retry_count >= EG800_NETWORK_RETRY_MAX) {
                network_info.state = EG800_NETWORK_STATE_FAULT;
                wait_ticks = portMAX_DELAY;
                LOG_ERROR("Network 本轮 %u 次联网均未成功，停止自动重试", (unsigned int)s_eg800_network_retry_count);
            } else {
                network_info.state = EG800_NETWORK_STATE_RETRY_WAIT;
                retry_start_tick = xTaskGetTickCount();
                wait_ticks = retry_interval_ticks;
                LOG_WARN("Network 第 %u 次联网未成功，PDP 已清理，等待 %lu ms 后重试", (unsigned int)s_eg800_network_retry_count, (unsigned long)EG800_NETWORK_RETRY_INTERVAL_MS);
            }

            Eg800_Network_Update_Snapshot(&network_info);
            continue;
        }
    }
}
